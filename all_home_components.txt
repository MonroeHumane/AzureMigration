<script>
  function initAccomplishmentsTabs() {
    const widget = document.querySelector<HTMLElement>('[data-monroe-accomplishments-widget]');
    if (!widget) return;
    
    const tabs = widget.querySelectorAll<HTMLButtonElement>('[role="tab"]');
    const panels = widget.querySelectorAll<HTMLElement>('[role="tabpanel"]');
    
    tabs.forEach((tab) => {
      tab.addEventListener('click', () => {
        const targetId = tab.getAttribute('aria-controls');
        
        // Deactivate all
        tabs.forEach(t => {
          t.setAttribute('aria-selected', 'false');
          t.setAttribute('tabindex', '-1');
        });
        panels.forEach(p => {
          p.hidden = true;
          p.classList.replace('opacity-100', 'opacity-0');
          p.classList.replace('z-10', 'z-0');
          p.classList.add('pointer-events-none');
        });
        
        // Activate target
        tab.setAttribute('aria-selected', 'true');
        tab.setAttribute('tabindex', '0');
        
        const targetPanel = widget.querySelector<HTMLElement>(`#${targetId}`);
        if (targetPanel) {
          targetPanel.hidden = false;
          // Small timeout to allow display:block to apply before opacity transition
          setTimeout(() => {
            targetPanel.classList.replace('opacity-0', 'opacity-100');
            targetPanel.classList.replace('z-0', 'z-10');
            targetPanel.classList.remove('pointer-events-none');
          }, 10);
        }
      });
    });
  }
  
  document.addEventListener('astro:page-load', initAccomplishmentsTabs);
  if (document.readyState === 'complete') {
    initAccomplishmentsTabs();
  } else {
    document.addEventListener('DOMContentLoaded', initAccomplishmentsTabs);
  }
</script>

  function initContactForm() {
    const form = document.querySelector<HTMLFormElement>('[data-monroe-contact-mailto]');
    const feedback = document.querySelector<HTMLElement>('.monroe-contact-form__feedback');
    if (!form || !feedback) return;

    const emailTo = form.dataset.email || 'info@monroe-humane.org';

    form.addEventListener('submit', (e) => {
      e.preventDefault();
      const hp = (document.getElementById('monroe_contact_website') as HTMLInputElement)?.value;
      if (hp) {
        feedback.textContent = 'Submission blocked (spam detected).';
        feedback.classList.remove('hidden');
        return;
      }
      
      const name = (document.getElementById('monroe_contact_name') as HTMLInputElement)?.value || '';
      const phone = (document.getElementById('monroe_contact_phone') as HTMLInputElement)?.value || '';
      const email = (document.getElementById('monroe_contact_email') as HTMLInputElement)?.value || '';
      const message = (document.getElementById('monroe_contact_message') as HTMLTextAreaElement)?.value || '';
      
      let body = `Name: ${name}\n`;
      if (phone) body += `Phone: ${phone}\n`;
      body += `Email: ${email}\n\n`;
      body += `Message:\n${message}\n`;
      
      const subject = `Website Contact Form: ${name}`;
      const mailtoUrl = `mailto:${encodeURIComponent(emailTo)}?subject=${encodeURIComponent(subject)}&body=${encodeURIComponent(body)}`;
      
      window.location.href = mailtoUrl;
      feedback.textContent = 'Opening your email client to complete the message...';
      feedback.classList.remove('hidden');
    });
  }

  document.addEventListener('astro:page-load', initContactForm);
  if (document.readyState === 'complete') {
    initContactForm();
  } else {
    document.addEventListener('DOMContentLoaded', initContactForm);
  }
</script>
